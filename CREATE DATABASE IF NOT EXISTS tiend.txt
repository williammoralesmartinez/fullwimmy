CREATE DATABASE IF NOT EXISTS tienda_naturista;
USE tienda_naturista;


CREATE TABLE categorias (id_categoria INT AUTO INCREMENT PRIMARY KEY,
 nombre VARCHAR(100) NOT NULL,
descripcion TEXT) ENGINE=InnoDB;

CREATE TABLE productos(id_producto INT AUTO INCREMEN PRIMARY KEY,
nombre VARCHAR(150) NOT NULL,
id_categoria INT,
precio DECIMAL(10,2) NOT NULL,
stock INT DEFAULT 0,
fecha_vencimiento DATE,
FOREIGN KEY (id_categoria) REFERENCES categorias(id_categoria)
ON DELETE SET NULL ON UPDATE CASCADE
)ENGINE=InnoDB;



CREATE TABLE clientes (id_cliente INT AUTO_INCREMENT PRIMARY KEY,
nombre VARCHAR(100) NOT NULL,
apellido VARCHAR(100) NOT NULL,
email  VARCHAR(150)UNIQUE,
telefono VARCHAR(20),
direccion TEXT
)ENGINE=InnoDB;

CREATE TABLE pedidos(id_pedido INT AUTO_INCREMENT PRIMARY KEY,
id_cliente INT,
fecha_pedido DATETIME DEFAULT CURREND_TIMESTAMP,
total DECIMAL(10, 2) DEFAU 0.00,
metodo_pago ENUM ('EFECTIVO','TARJETA','TRANSFERENCIA'),
FOREING KEY (id_cliente) REFERENCES clientes(id_cliente)
ON DELETE CASCADE ON UPDATE CASCADE) ENGINE=InnoDB;

CREATE TABLE detalle_pedidos(id_detalle INT AUTO_INCREMENT PRIMARY KEY,
id_pedido INT,
id_producto INT,
cantidad  INT NOT NULL,
precio_unitario DECIMAL(10,2) NOT NULL,
subtotal DECIMAL (10,2) AS (cantidad * precio_unitario),
FOREIGN KEY (id_pedido) REFERENCES pedidos(id_pedido) ON DELETE CASCADE,
FOREIGN KEY (id_producto) REFERENCES productos(id_producto)ON DELETE RESTRICT
)ENGINE=InnoDB;
